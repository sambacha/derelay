func TestMetricsCounters(t *testing.T) {
	is := is.New(t)

	// Reset metrics (assuming they are global)
	// Ideally, we'd create a new registry for testing, but work with globals for now.
	// This might require exposing a reset function or careful test ordering.
	// For simplicity, we test increments from the current state.

	// Get initial values (or assume zero if we could reset) - Use Exported variables
	initialConns := testutil.ToFloat64(metrics.CurrentConnections) // Use Exported names
	initialNew := testutil.ToFloat64(metrics.NewConnections)
	initialClosed := testutil.ToFloat64(metrics.ClosedConnections)
	initialMsgRecv := testutil.ToFloat64(metrics.MessagesReceived)
	initialMsgSent := testutil.ToFloat64(metrics.MessagesSent)
	initialCached := testutil.ToFloat64(metrics.CachedMessages)
	initialNewSess := testutil.ToFloat64(metrics.NewRequestedSessions)
	initialRecvSess := testutil.ToFloat64(metrics.ReceivedSessions)
	initialEstSess := testutil.ToFloat64(metrics.EstablishedSessions)

	// Test connection metrics - Use exported functions
	metrics.IncNewConnection()        // Use Exported func
	metrics.SetCurrentConnections(10) // Use Exported func

	is.Equal(testutil.ToFloat64(metrics.NewConnections), initialNew+1)    // Use Exported var
	is.Equal(testutil.ToFloat64(metrics.CurrentConnections), float64(10)) // Use Exported var

	metrics.IncClosedConnection()                                            // Use Exported func
	is.Equal(testutil.ToFloat64(metrics.ClosedConnections), initialClosed+1) // Use Exported var

	// Test message metrics - Use exported functions
	metrics.IncMessagesReceived()      // Use Exported func (renamed from IncTotalMessages)
	metrics.IncMessagesSentFromCache() // Use Exported func (renamed from DecCachedMessages)
	metrics.IncCachedMessages()        // Use Exported func

	is.Equal(testutil.ToFloat64(metrics.MessagesReceived), initialMsgRecv+1) // Use Exported var
	is.Equal(testutil.ToFloat64(metrics.MessagesSent), initialMsgSent+1)     // Use Exported var (check total sent)
	is.Equal(testutil.ToFloat64(metrics.CachedMessages), initialCached+1)    // Use Exported var

	// Test session metrics - Use exported functions
	metrics.IncNewRequestedSessions() // Use Exported func
	metrics.IncReceivedSessions()     // Use Exported func
	metrics.IncEstablishedSessions()  // Use Exported func

	is.Equal(testutil.ToFloat64(metrics.NewRequestedSessions), initialNewSess+1) // Use Exported var
	is.Equal(testutil.ToFloat64(metrics.ReceivedSessions), initialRecvSess+1)    // Use Exported var
	is.Equal(testutil.ToFloat64(metrics.EstablishedSessions), initialEstSess+1)  // Use Exported var
}
